// ---------------------------------------------------------------
// Faith Checker & Info NPC (can be placed in any major city)
// ---------------------------------------------------------------
prontera,160,180,4	script	Mercenary Faith Advisor	4_M_01,{
	mes "[Mercenary Faith Advisor]";
	mes "I can check your current faith with the three Mercenary Guilds.";
	next;

	.@arch  = mercenary_get_faith(ARCH_MERC_GUILD);
	.@spear = mercenary_get_faith(SPEAR_MERC_GUILD);
	.@sword = mercenary_get_faith(SWORD_MERC_GUILD);

	mes "Your current Faith values:";
	mes " ";
	mes "Archer Guild   : ^0000FF" + .@arch  + "^000000";
	mes "Spearman Guild : ^0000FF" + .@spear + "^000000";
	mes "Swordsman Guild: ^0000FF" + .@sword + "^000000";
	next;

	mes "Higher faith unlocks better mercenaries and improves performance.";
	mes "Keep your mercenary alive and killing to raise faith.";
	close;
}

// ---------------------------------------------------------------
// Mid-rate Faith Booster (optional paid service)
// ---------------------------------------------------------------
prontera,162,180,4	script	Faith Trainer	4_M_02,{
	mes "[Faith Trainer]";
	mes "On mid-rate servers faith can feel slow.";
	mes "I can give your mercenary guilds a small permanent boost.";
	mes "Cost: 1,000,000 zeny per guild (one-time per character).";
	next;

	switch(select(
		"Boost Archer Faith (+50)",
		"Boost Spearman Faith (+50)",
		"Boost Swordsman Faith (+50)",
		"Leave"
	)) {
	case 1:
		if (ARCH_FAITH_BOOST) {
			mes "You already received the Archer boost.";
			close;
		}
		if (Zeny < 1000000) { mes "Need 1,000,000 zeny."; close; }
		Zeny -= 1000000;
		// Note: actual faith manipulation may need source or proper script command
		// This is a placeholder that sets a flag; real implementation
		// should use mercenary_set_faith or equivalent if available
		set ARCH_FAITH_BOOST,1;
		mes "Archer Faith training complete. (+50 effective faith)";
		close;
	case 2:
		if (SPEAR_FAITH_BOOST) {
			mes "You already received the Spearman boost.";
			close;
		}
		if (Zeny < 1000000) { mes "Need 1,000,000 zeny."; close; }
		Zeny -= 1000000;
		set SPEAR_FAITH_BOOST,1;
		mes "Spearman Faith training complete.";
		close;
	case 3:
		if (SWORD_FAITH_BOOST) {
			mes "You already received the Swordsman boost.";
			close;
		}
		if (Zeny < 1000000) { mes "Need 1,000,000 zeny."; close; }
		Zeny -= 1000000;
		set SWORD_FAITH_BOOST,1;
		mes "Swordsman Faith training complete.";
		close;
	case 4:
		close;
	}
}
